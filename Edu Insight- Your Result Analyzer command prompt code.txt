import pandas as pd

import seaborn as sns

# List of semester files and corresponding sheet names
semester_files = {
    'Semester 1': 'sem1_student_results.xlsx',
    'Semester 2': 'sem2_student_results.xlsx',
    'Semester 3': 'sem3_student_results.xlsx',
    'Semester 4': 'sem4_student_results.xlsx'
}

# Initialize dictionaries to hold data from each semester
marks_data_all = {}
subject_data_all = {}

# Function for calculating result class
def calculate_result_class(row, semester):
    if semester in ['Semester 1', 'Semester 2']:
        if row['Total marks'] >= 640: return 'Distinction'
        elif row['Total marks'] >= 480: return 'First Class'
        elif row['Total marks'] >= 320: return 'Second Class'
        elif row['Total marks'] >= 224: return 'Pass'
        else: return 'Fail'
    elif semester in ['Semester 3', 'Semester 4']:
        if row['Total marks'] >= 720: return 'Distinction'
        elif row['Total marks'] >= 540: return 'First Class'
        elif row['Total marks'] >= 360: return 'Second Class'
        elif row['Total marks'] >= 252: return 'Pass'
        else: return 'Fail'

# Loop through each semester file and load the sheets
for semester, file in semester_files.items():
    marks_data_all[semester] = pd.read_excel(file, sheet_name='Student Marks')
    subject_data_all[semester] = pd.read_excel(file, sheet_name='Subject Details')

    # Convert subject columns to numeric and fill missing values with 0
    marks_data_all[semester].iloc[:, 2:-2] = marks_data_all[semester].iloc[:, 2:-2].apply(pd.to_numeric, errors='coerce')
    marks_data_all[semester].iloc[:, 2:-2].fillna(0, inplace=True)

    # Calculate Total Marks
    marks_data_all[semester]['Total marks'] = marks_data_all[semester].iloc[:, 2:-2].sum(axis=1)

    # Calculate Result Class
    marks_data_all[semester]['Result Class'] = marks_data_all[semester].apply(lambda row: calculate_result_class(row, semester), axis=1)

import matplotlib.pyplot as plt
import pandas as pd
import numpy as np
def visualize_student_performance(usn):
    """
    Visualize a student's performance across all semesters and overall.
    :param usn: The unique student number (USN) of the student.
    """
    # Initialize structures to hold aggregated data
    semester_performance = {}
    all_subjects = set()
    total_marks_across_sems = []
    pass_fail_distribution = {'Passed': 0, 'Failed': 0}

    for semester, data in marks_data_all.items():
        # Filter data for the given USN
        student_row = data[data['USN'] == usn]

        if student_row.empty:
            print(f"Student with USN {usn} not found in {semester}.")
            continue

        # Collect all unique subject names
        subjects = data.columns[2:-2].tolist()
        all_subjects.update(subjects)

        # Collect semester-specific data
        semester_performance[semester] = {
            'Total Marks': student_row['Total marks'].values[0],
            'Result Class': student_row['Result Class'].values[0],
            'Subject Marks': dict(zip(subjects, student_row.iloc[:, 2:-2].values.flatten())),
        }

        # Update overall performance data
        total_marks_across_sems.append(student_row['Total marks'].values[0])
        pass_fail_distribution['Passed'] += (student_row.iloc[:, 2:-2] > 35).sum(axis=1).values[0]
        pass_fail_distribution['Failed'] += (student_row.iloc[:, 2:-2] <= 35).sum(axis=1).values[0]

    # If no data is found for the USN
    if not semester_performance:
        print(f"No data found for USN: {usn} in any semester.")
        return

    # Convert all_subjects to a sorted list for consistent ordering
    all_subjects = sorted(all_subjects)

    # Prepare data for visualization
    subject_marks_by_semester = {subject: [] for subject in all_subjects}
    for semester, performance in semester_performance.items():
        for subject in all_subjects:
            subject_marks_by_semester[subject].append(performance['Subject Marks'].get(subject, 0))

    # Visualize Marks Across Semesters
    plt.figure(figsize=(10, 6))
    for semester, performance in semester_performance.items():
        plt.bar(semester, performance['Total Marks'], label=semester)
    plt.title(f'Total Marks Across Semesters - USN: {usn}')
    plt.ylabel('Total Marks')
    plt.xlabel('Semesters')
    plt.legend()
    plt.show()

    # Visualize Subject-Wise Performance Across Semesters
    bar_width = 0.2
    x_indices = np.arange(len(all_subjects))
    plt.figure(figsize=(12, 8))
    for i, semester in enumerate(semester_performance.keys()):
        marks = [subject_marks_by_semester[subject][i] if i < len(subject_marks_by_semester[subject]) else 0 for subject in all_subjects]
        plt.bar(x_indices + i * bar_width, marks, width=bar_width, label=semester)
    plt.title(f'Subject-wise Performance Across Semesters - USN: {usn}')
    plt.ylabel('Marks')
    plt.xticks(x_indices + bar_width / 2, all_subjects, rotation=45)
    plt.legend()
    plt.tight_layout()
    plt.show()

    # Visualize Overall Pass/Fail Distribution
    plt.figure(figsize=(8, 6))
    labels = ['Passed Subjects', 'Failed Subjects']
    sizes = [pass_fail_distribution['Passed'], pass_fail_distribution['Failed']]
    colors = ['green', 'red']
    plt.pie(sizes, labels=labels, autopct='%1.1f%%', startangle=140, colors=colors)
    plt.title(f'Overall Pass/Fail Distribution - USN: {usn}')
    plt.axis('equal')  # Equal aspect ratio ensures the pie chart is circular.
    plt.show()

    # Print Overall Summary
    print(f"Overall Performance Summary for USN: {usn}")
    print("===============================================")
    print(f"Total Marks Across Semesters: {sum(total_marks_across_sems)}")
    print(f"Average Marks Across Semesters: {sum(total_marks_across_sems) / len(total_marks_across_sems):.2f}")
    print(f"Overall Pass Percentage: {100 * pass_fail_distribution['Passed'] / (pass_fail_distribution['Passed'] + pass_fail_distribution['Failed']):.2f}%")

usn = input("Enter the student's USN: ")
visualize_student_performance(usn)
